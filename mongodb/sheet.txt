///// Insert (Create)
db.collection_name.inserOne({}) - single doc Insert
db.collection_name.insertMany([{},{},{}]) - Multiple doc Insert

///// Find(Read)
db.collection_name.find()
db.collection_name.findOne()
db.collection_name.find({"name":"devanshu"})
db.collection_name.findOne({"name":"devanshu"})

//// Delete
db.collection_name.deleteOne({"name":"devanshu"}) - single doc delete (first find)
db.collection_name.deleteMany({"name":"devanshu"}) - multiple doc delete 

//// Update 
db.collection_name.updateOne({"name":"devanshu"},{$set:{"name":"manny"}}) - single doc update (first find)

db.collection_name.updateMany({"name":"devanshu"},{$set:{"name":"manny"}}) - multiple doc update
